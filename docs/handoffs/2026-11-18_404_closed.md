### Step 1: put the server and the payments feed on the new code

**Which computer:** the home desktop. **Which window:** the Ubuntu
terminal. Run it after the pull request is merged. It takes about two
minutes, and asks for your Ubuntu password once.

What it does, in order:
1. **Checks before changing anything.** The server and the payments feed
   must be running. The server's copy of the code must be clean and on
   master. GitHub's master must hold this phase, and live must already
   have the new table.
2. **Brings the copy up to date** (fast-forward only).
3. **Restarts the server** and waits for it to answer healthy.
4. **Restarts the payments feed** and waits for its first check-in to be
   written down.
5. **Prints the payments feed's status** and checks that Sentry accepted
   the check-in.

If the server or the feed does not come back, the code is put back and
both are restarted on it.

```bash
bash <<'EOF2'
set -u
bad() { echo; echo "SOMETHING IS OFF: $1"; exit 1; }
SRV=~/motodiag/server
LIVE=~/motodiag/live
ENVF=~/.config/motodiag/server.env
PY="$SRV/.venv/bin/python"
API=motodiag-api.service
FEED=motodiag-payments.service
FIX=af96fb973cab2a9387eedd190784d6a84e948e63
LOAD='import os, sys
from pathlib import Path
for k in [k for k in os.environ if k.startswith("MOTODIAG_")]:
    del os.environ[k]
os.environ["MOTODIAG_DB_PATH"] = sys.argv[2] + "/motodiag.db"
os.environ["MOTODIAG_DATA_DIR"] = sys.argv[2]
for raw in Path(sys.argv[1]).read_text(encoding="utf-8").splitlines():
    line = raw.strip()
    if not line or line[0] in "#;" or "=" not in line:
        continue
    k, v = (p.strip() for p in line.split("=", 1))
    if len(v) > 1 and v[0] == v[-1] and v[0] in "\"\x27":
        v = v[1:-1]
    os.environ[k] = v
'
STATUS="$LOAD"'os.execv(sys.argv[3], [sys.argv[3], "payments", "feed", "status"])'
rows() { python3 -c '
import sqlite3, sys
db = sqlite3.connect("file:" + sys.argv[1] + "?mode=ro", uri=True)
t = db.execute("SELECT 1 FROM sqlite_master WHERE type = ? AND name = ?", ("table", "stripe_feed_checkins")).fetchone()
print(db.execute("SELECT COUNT(*) FROM stripe_feed_checkins").fetchone()[0] if t else "none")' "$LIVE/motodiag.db" </dev/null; }
mainpid() { systemctl show -p MainPID --value "$FEED" </dev/null; }
healthy() {
  for i in $(seq 30); do
    curl -s --max-time 5 http://127.0.0.1:8000/healthz </dev/null | grep -q '"status":"ok"' && return 0
    sleep 2
  done
  return 1
}

echo "Checking the server, the payments feed and live..."
for T in git systemctl sudo curl python3 sleep grep sed seq; do
  command -v "$T" >/dev/null 2>&1 || bad "this computer has no $T command; nothing was changed, so bring this to the advisor"
done
{ [ -d "$SRV/.git" ] && [ -x "$PY" ] && [ -f "$LIVE/motodiag.db" ] && [ -f "$ENVF" ]; } || bad "this is not the computer the server runs on (run it on the home desktop, in the Ubuntu window); nothing was changed"
systemctl is-active -q "$API" </dev/null || bad "the server is not running right now; nothing was changed, so bring this to the advisor"
systemctl is-active -q "$FEED" </dev/null || bad "the payments feed is not running right now; nothing was changed, so bring this to the advisor"
BEFORE_ROWS=$(rows) || bad "live could not be read; nothing was changed, so bring this to the advisor"
[ "$BEFORE_ROWS" != none ] || bad "live does not have the new table for check-ins yet; nothing was changed, so bring this to the advisor"
[ "$(git -C "$SRV" branch --show-current </dev/null)" = master ] || bad "the server's copy of the code is not on master; nothing was changed, so bring this to the advisor"
[ -z "$(git -C "$SRV" status --porcelain --untracked-files=no </dev/null)" ] || bad "the server's copy of the code has changes made by hand; nothing was changed, so bring this to the advisor"
BEFORE=$(git -C "$SRV" rev-parse HEAD </dev/null)
git -C "$SRV" fetch -q origin +refs/heads/master:refs/remotes/origin/master </dev/null || bad "could not fetch from GitHub; nothing was changed"
TARGET=$(git -C "$SRV" rev-parse origin/master </dev/null)
git -C "$SRV" merge-base --is-ancestor "$FIX" "$TARGET" </dev/null || bad "master on GitHub does not have this phase yet: merge the pull request first, then paste this block again; nothing was changed"
git -C "$SRV" diff --quiet "$FIX" "$TARGET" -- src pyproject.toml </dev/null || bad "master has new code since this phase that has not been checked here; nothing was changed, so bring this to the advisor"
git -C "$SRV" merge-base --is-ancestor "$BEFORE" "$TARGET" </dev/null || bad "the server's copy of the code has commits that are not on master; nothing was changed, so bring this to the advisor"
if [ "$BEFORE" = "$TARGET" ]; then
  echo "The server's copy of the code is already up to date."
else
  echo "Bringing the server's copy of the code up to date..."
  git -C "$SRV" merge -q --ff-only "$TARGET" </dev/null || bad "could not bring the server's copy of the code up to date; nothing was changed"
fi

echo
echo "Your Ubuntu password, once, to restart the server and the payments feed (only the code has changed so far):"
sudo -v </dev/null || { git -C "$SRV" reset -q --keep "$BEFORE" </dev/null; bad "sudo did not take the password; the code was put back as it was and nothing was restarted"; }
putback() {
  echo "Putting the server and the payments feed back on the old code..."
  git -C "$SRV" reset -q --keep "$BEFORE" </dev/null
  sudo systemctl restart "$API" </dev/null
  sudo systemctl restart "$FEED" </dev/null
  sleep 30
  if healthy && systemctl is-active -q "$FEED" </dev/null; then
    bad "$1. Both are back on the old code and running; bring this to the advisor"
  fi
  bad "$1, and the server or the payments feed is not running even on the old code: bring this to the advisor now"
}

echo "Restarting the server (up to 60 seconds)..."
sudo systemctl restart "$API" </dev/null || putback "the server could not be restarted"
healthy || putback "the server did not come back healthy on the new code"
echo "The server is healthy on the new code."

OLDPID=$(mainpid)
echo "Restarting the payments feed, and waiting for its first check-in (up to 3 minutes)..."
sudo systemctl restart "$FEED" </dev/null || putback "the payments feed could not be restarted"
NOW_ROWS="$BEFORE_ROWS"
for i in $(seq 36); do
  sleep 5
  NOW_ROWS=$(rows) || NOW_ROWS="$BEFORE_ROWS"
  [ "$NOW_ROWS" != "$BEFORE_ROWS" ] && break
done
NEWPID=$(mainpid)
systemctl is-active -q "$FEED" </dev/null || putback "the payments feed did not keep running on the new code"
{ [ -n "$NEWPID" ] && [ "$NEWPID" != 0 ] && [ "$NEWPID" != "$OLDPID" ]; } || putback "the payments feed did not actually restart"
[ "$NOW_ROWS" != "$BEFORE_ROWS" ] || bad "the payments feed runs the new code but wrote down no check-in in 3 minutes; it asks Stripe as before, so bring this to the advisor"
echo "The payments feed wrote down its first check-in on the new code."

echo "The payments feed says:"
SAYS=$(cd "$SRV" && "$PY" -c "$STATUS" "$ENVF" "$LIVE" "$SRV/.venv/bin/motodiag" </dev/null 2>&1); RC=$?
printf '%s\n' "$SAYS" | sed 's/^/    /'
printf '%s\n' "$SAYS" | grep -q "^Last check-in: .*Sentry accepted it" || bad "the status does not show Sentry accepting the last check-in (its lines above say why); both run the new code, so bring this to the advisor"
[ "$RC" = 0 ] || bad "the status says something is not right (its lines above say why); both run the new code, so bring this to the advisor"
echo "The server and the payments feed are on: $(git -C "$SRV" log -1 --format='%h %s' </dev/null)"
echo "ALL GOOD"
EOF2
```
