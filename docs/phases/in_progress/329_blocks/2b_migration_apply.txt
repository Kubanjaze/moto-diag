set -u
bad() { echo; echo "SOMETHING IS OFF: $1"; exit 1; }
BRANCH=phase-329-self-serve-signup
REPO=~/Projects/moto-diag
PY="$REPO/.venv/bin/python"
RM="$REPO/.claude/skills/deploy/remote.py"
{ [ -x "$PY" ] && [ -f "$RM" ]; } || bad "this is not the Mac with the project (run it on the Mac, in Terminal); nothing was changed"

[ -z "$(git status --porcelain)" ] || bad "the project folder on this Mac has changes that are not committed; nothing was changed"

git checkout -q "$BRANCH" < /dev/null && git merge -q --ff-only "origin/$BRANCH" < /dev/null || bad "could not put this Mac on the sign-up branch; nothing was changed"

  "$PY" "$RM" api-start < /dev/null
  bad "the apply refused or failed (the lines above say why); the server was started again"
fi
echo "3 of 5: the app server started again"
"$PY" "$RM" api-start < /dev/null || bad "the change is in, but the server would not start: paste this output in the project thread now"
echo "4 of 5: live checked against the approved list"
"$PY" "$RM" verify-live 329 < /dev/null || bad "live does not hold the approved change (the lines above say why)"
