bad() { echo; echo "SOMETHING IS OFF: $1"; exit 1; }
BRANCH=phase-329-self-serve-signup

F=docs/phases/in_progress/329_dryrun_diff.md

  grep -q "schema added: $t" "$F" || bad "the list does not add $t"
