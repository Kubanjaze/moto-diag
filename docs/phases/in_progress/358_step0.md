```python
"""Phase 262 deploy, in two separate steps (264's pattern).

  deploy262.py dryrun      before-state from live (read only), backup (SQLite
                           backup API, retain 5), migrate a COPY of the backup,
                           diff every table by rowid, print every changed row's
                           before and after text, run the F158 census on the
                           copy, and check the scope. Live is only read.
  deploy262.py apply-live  only after the operator's answer: prove live equals
                           the backup, re-run the migration on a fresh copy of
                           the backup with the code about to run live and
                           check the scope, then migrate live, diff live
                           against the backup, same scope check.

Scope (the rule-1 stop): workflow_templates 0 changed + 3 added;
checklist_items changed = exactly the rows in SCOPE, each only in its named
fields, + 21 added; schema_version +1; nothing else changed, added or removed.
"""
import datetime as dt, hashlib, json, pathlib, sqlite3, subprocess, sys
REPO = pathlib.Path("/Users/lilquant/Projects/moto-diag")
LIVE = REPO / "data" / "motodiag.db"
BK = pathlib.Path.home() / "backups" / "motodiag"
HERE = pathlib.Path(__file__).parent
sys.path.insert(0, str(REPO / "src"))

# (template slug, how the row is found, the fields allowed to change)
SCOPE = [
    ("generic_winterization_v1", ("sequence_number", 1), {"instruction_text", "expected_pass", "expected_fail"}),
    ("generic_winterization_v1", ("sequence_number", 2), {"instruction_text"}),
    ("generic_winterization_v1", ("sequence_number", 3), {"expected_pass"}),
    ("ppi_chassis_v1", ("title", "Steering head bearings"), {"description", "instruction_text"}),
    ("winterization_v1", ("sequence_number", 5), {"description"}),
]
ADDED_TEMPLATES, ADDED_ITEMS = 3, 21

def state(path):
    c = sqlite3.connect(f"file:{path}?mode=ro", uri=True)
    ki = hashlib.sha256()
    for r in c.execute("SELECT * FROM known_issues ORDER BY id"):
        ki.update(repr(r).encode())
    s = {"schema": c.execute("SELECT MAX(version) FROM schema_version").fetchone()[0],
         "known_issues": c.execute("SELECT COUNT(*) FROM known_issues").fetchone()[0],
         "templates": c.execute("SELECT COUNT(*) FROM workflow_templates").fetchone()[0],
         "items": c.execute("SELECT COUNT(*) FROM checklist_items").fetchone()[0],
         "integrity": c.execute("PRAGMA integrity_check").fetchone()[0],
         "known_issues_sha256": ki.hexdigest()[:16]}
    c.close(); return s

def tables(c):
    return [r[0] for r in c.execute("SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%'")]

def dump(path):
    c = sqlite3.connect(f"file:{path}?mode=ro", uri=True); out = {}
    for t in tables(c):
        cols = [r[1] for r in c.execute(f"PRAGMA table_info('{t}')")]
        out[t] = (cols, {r[0]: r for r in c.execute(f"SELECT rowid, * FROM '{t}'")})
    c.close(); return out

def diff(a, b):
    res = {}
    for t in sorted(set(a) | set(b)):
        cols, ra = a.get(t, ([], {})); _, rb = b.get(t, ([], {}))
        added = sorted(set(rb) - set(ra)); removed = sorted(set(ra) - set(rb))
        changed = sorted(k for k in set(ra) & set(rb) if ra[k] != rb[k])
        if added or removed or changed:
            res[t] = {"cols": cols, "added": added, "removed": removed, "changed": changed, "a": ra, "b": rb}
    return res

def scoped_rowids(path):
    """rowid -> allowed fields, resolved on the BEFORE database."""
    c = sqlite3.connect(f"file:{path}?mode=ro", uri=True); out = {}
    for slug, (col, val), fields in SCOPE:
        rows = c.execute(f"SELECT i.rowid FROM checklist_items i JOIN workflow_templates t ON t.id = i.template_id "
                         f"WHERE t.slug = ? AND i.{col} = ?", (slug, val)).fetchall()
        if len(rows) != 1: raise SystemExit(f"scope row {slug} {col}={val!r}: {len(rows)} rows, not 1")
        out[rows[0][0]] = fields
    c.close(); return out

def check_scope(d, allowed):
    probs = []
    for t, x in d.items():
        if t not in {"workflow_templates", "checklist_items", "schema_version"}: probs.append(f"{t}: touched")
        if x["removed"]: probs.append(f"{t}: {len(x['removed'])} removed")
    wt, ci, sv = d.get("workflow_templates"), d.get("checklist_items"), d.get("schema_version")
    if not wt or wt["changed"] or len(wt["added"]) != ADDED_TEMPLATES:
        probs.append(f"workflow_templates: changed {wt and wt['changed']}, added {wt and len(wt['added'])}")
    if not ci:
        probs.append("checklist_items untouched")
    else:
        if set(ci["changed"]) != set(allowed): probs.append(f"checklist_items changed rowids {ci['changed']}, not {sorted(allowed)}")
        if len(ci["added"]) != ADDED_ITEMS: probs.append(f"checklist_items +{len(ci['added'])}")
        cols = ["rowid"] + ci["cols"]
        for k in ci["changed"]:
            moved = {cols[i] for i in range(len(cols)) if ci["a"][k][i] != ci["b"][k][i]}
            if moved != allowed.get(k, set()): probs.append(f"checklist_items rowid {k} changed {sorted(moved)}, allowed {sorted(allowed.get(k, set()))}")
    if not sv or len(sv["added"]) != 1 or sv["changed"]: probs.append("schema_version not exactly +1")
    return probs

def report(d, out):
    L = []
    for t, x in d.items():
        L.append(f"## {t}: +{len(x['added'])} added, {len(x['changed'])} changed, {len(x['removed'])} removed\n")
        for k in x["changed"]:
            L.append(f"\n### {t} rowid {k}\n")
            for i, col in enumerate(["rowid"] + x["cols"]):
                va, vb = x["a"][k][i], x["b"][k][i]
                if va != vb: L.append(f"\n**{col}**\n\n- before: {va}\n- after:  {vb}\n")
        if x["added"] and t != "checklist_items":
            for k in x["added"]: L.append(f"- added rowid {k}: {x['b'][k][1:4]}\n")
        if t == "checklist_items" and x["added"]:
            L.append(f"- added rowids {x['added'][0]}–{x['added'][-1]} ({len(x['added'])} items)\n")
    out.write_text("".join(L)); return "".join(L)

def backup(src, dst):
    s = sqlite3.connect(f"file:{src}?mode=ro", uri=True); d = sqlite3.connect(dst); s.backup(d); s.close(); d.close()

def retain5():
    dbs = sorted(BK.glob("motodiag_*.db"), key=lambda p: p.stat().st_mtime, reverse=True); removed = []
    for p in dbs[5:]:
        for q in (p, pathlib.Path(str(p) + "-shm"), pathlib.Path(str(p) + "-wal")):
            if q.exists(): q.unlink()
        removed.append(p.name)
    return removed

def migrate(path):
    from motodiag.core.migrations import apply_pending_migrations
    return apply_pending_migrations(str(path))

def census(path):
    return subprocess.run([sys.executable, str(HERE / "f158.py"), str(path)], capture_output=True, text=True).stdout

if __name__ == "__main__":
    mode = sys.argv[1]
    if mode == "dryrun":
        before = state(LIVE); print("before (live, read only):", json.dumps(before))
        stamp = dt.datetime.now().strftime("%Y%m%d_%H%M%S")
        bk = BK / f"motodiag_pre262_{stamp}.db"; backup(LIVE, bk)
        print("backup:", bk, json.dumps(state(bk)))
        print("retain 5 removed:", retain5())
        copy = HERE / "dryrun_copy.db"
        for q in (copy, pathlib.Path(str(copy) + "-wal"), pathlib.Path(str(copy) + "-shm")):
            if q.exists(): q.unlink()
        backup(bk, copy)
        allowed = scoped_rowids(copy)
        print("applied on copy:", migrate(copy))
        d = diff(dump(bk), dump(copy)); report(d, HERE / "dryrun_diff.md")
        print("after (copy):", json.dumps(state(copy)))
        print("scope problems:", check_scope(d, allowed) or "none")
        print("F158 census on the copy:\n" + census(copy))
        (HERE / "dryrun_backup.txt").write_text(str(bk))
    elif mode == "apply-live":
        bk = pathlib.Path((HERE / "dryrun_backup.txt").read_text().strip())
        drift = diff(dump(bk), dump(LIVE))
        if drift: sys.exit(f"REFUSED: live differs from the backup in {sorted(drift)}; stop and ask")
        print("live == backup, every table, every row")
        pre = HERE / "preapply_copy.db"
        for q in (pre, pathlib.Path(str(pre) + "-wal"), pathlib.Path(str(pre) + "-shm")):
            if q.exists(): q.unlink()
        backup(bk, pre); allowed = scoped_rowids(pre)

        probs = check_scope(diff(dump(bk), dump(pre)), allowed)
        if probs: sys.exit(f"REFUSED: pre-apply dry run out of scope: {probs}; stop and ask")
        print("pre-apply scope problems: none")
        print("applied live:", migrate(LIVE))
        d = diff(dump(bk), dump(LIVE)); report(d, HERE / "live_diff.md")
        print("after (live):", json.dumps(state(LIVE)))
        print("scope problems:", check_scope(d, allowed) or "none")
        print("F158 census on live:\n" + census(LIVE))
```

```python
"""F158 census: every text column of every table, the operator's three
patterns plus "this phase". usage: f158.py DB [--plant]  (--plant works on a
copy with "Phase 999" planted in one known_issues.description)."""
import collections, re, shutil, sqlite3, sys
PATS = {"Phase N": re.compile(r"\bPhase \d+"), "Track X": re.compile(r"\bTrack [A-Z]\b"),
        "F-number": re.compile(r"\bF\d{2,3}\b"), "this phase": re.compile(r"this phase", re.I)}
def census(path):
    c = sqlite3.connect(path); hits = []
    for (t,) in c.execute("select name from sqlite_master where type='table' and name not like 'sqlite_%'"):
        cols = [r[1] for r in c.execute(f'pragma table_info("{t}")') if (r[2] or "").upper() in ("TEXT", "") or "CHAR" in (r[2] or "").upper()]
        for col in cols:
            for rowid, v in c.execute(f'select rowid, "{col}" from "{t}" where "{col}" is not null'):
                if isinstance(v, str):
                    for name, p in PATS.items():
                        for m in p.finditer(v): hits.append((t, col, rowid, name, m.group()))
    c.close(); return hits
def report(hits):
    print("total hits", len(hits), "in", len({(t, r) for t, _, r, _, _ in hits}), "rows")
    for k, v in sorted(collections.Counter((t, c, n) for t, c, _, n, _ in hits).items()): print("  ", k, v)
    print("  F-number tokens:", sorted(collections.Counter(g for *_, n, g in hits if n == "F-number").items()))
    wf = [h for h in hits if h[0] in ("workflow_templates", "checklist_items")]
    print("  workflow rows:", wf)
if __name__ == "__main__":
    db = sys.argv[1]
    if "--plant" in sys.argv:
        cp = db + ".plant.db"; shutil.copy(db, cp); c = sqlite3.connect(cp)
        rid = c.execute("select min(rowid) from known_issues").fetchone()[0]
        c.execute("update known_issues set description = description || ' Phase 999' where rowid=?", (rid,)); c.commit(); c.close()
        base = census(db); planted = census(cp)
        extra = [h for h in planted if h not in base]
        print("plant control: extra hits", extra); sys.exit(0 if len(extra) == 1 and extra[0][4] == "Phase 999" else 1)
    report(census(db))
```
