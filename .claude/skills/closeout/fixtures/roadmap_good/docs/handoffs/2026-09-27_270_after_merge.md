```bash
gh issue edit 14 -R Kubanjaze/moto-diag --remove-label needs-admiral && echo "ALL GOOD"
gh pr view 3 --repo=Kubanjaze/moto-diag --json state && echo "ALL GOOD"
gh api -X PUT "repos/Kubanjaze/moto-diag/collaborators/$U" -f permission=push > /dev/null
L=$(GH_TOKEN="$T" gh api user --jq .login 2>/dev/null)
gh auth status >/dev/null 2>&1 || echo "SOMETHING IS OFF"

GH_REPO=Kubanjaze/moto-diag gh label list >/dev/null
```
